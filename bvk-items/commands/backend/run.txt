## 5. Docker Deployment & Orchestration

# Run the complete system locally (defaults to CLIENT_ID in .env, e.g. test_custom)
cd /Users/bvk/BVK_Workspace/BES
docker compose up --build -d

# Run the complete system for a specific client instance (e.g. acme_inc)
CLIENT_ID=acme_inc docker compose up --build -d

# Stop the complete system
docker compose down

# Build individual frontend container
cd /Users/bvk/BVK_Workspace/BES/bes-frontend
docker build -t bes-frontend-shell:latest .

# Build individual backend container for a specific client (e.g. acme_inc)
cd /Users/bvk/BVK_Workspace/BES/bes-backend
docker build -f instances/acme_inc/Dockerfile -t bes-backend-acme_inc:latest .
